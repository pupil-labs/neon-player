    def run_background_action(
    self, name: str, action_name: str, *args: object
) -> BackgroundJob:
    """Run an long-running job in background so that the UI stays responsive.

    Background jobs are not available when Neon Player is running in
    headless mode.

    Args:
        name: Job name shown in the UI.
        action_name: Name of the action to run, formatted as
            ``"PluginClass.method"``. The method should yield
            [`ProgressUpdate`][ProgressUpdate] values to report progress
            to the main application.
        *args: Arbitrary amount of action arguments.

    Returns:
        The running job.
    """
    app = neon_player.instance()
    if app.headless:
        raise RuntimeError("Not starting background job in headless mode")

    recording = app.recording
    app.save_settings(force=True)
    rec_dir = Path(str(recording._rec_dir)) if recording is not None else None

    job = BackgroundJob(
        name,
        self.job_counter,
        rec_dir,
        action_name,
        *args,
    )
    self.job_counter += 1

    job.canceled.connect(lambda: self.on_job_canceled(job))
    job.finished.connect(lambda: self.on_job_finished(job))
    job.progress_changed.connect(lambda _: self.updated.emit())

    self.current_jobs.append(job)
    self.job_started.emit(job)
    self.updated.emit()

    logger.info(f"{job.name} started in the background")

    return job

    def __init__(
    self,
    name: str,
    job_id: int,
    recording_path: Path | None,
    action_name: str,
    *args: object,
) -> None:
    super().__init__()

    self.name: str = name
    """Job name shown in the UI."""
    self.progress: float = -1
    """Last reported progress; ``-1`` until the first update."""

    self.job_id = job_id
    self.socket: QLocalSocket | None = None
    self._expected_payload_size: int | None = None

    server_name = f"neon-player-job-{job_id}"
    self.server = QLocalServer()
    QLocalServer.removeServer(server_name)  # remove any dangling sockets
    if not self.server.listen(server_name):
        logger.error(f"Could not start IPC server for job {name}")
        return

    self.server.newConnection.connect(self._handle_connection)

    command_args = [
        str(recording_path),
        "--progress-ipc-name",
        server_name,
        "--job",
        action_name,
    ] + [str(arg) for arg in args]

    if neon_player.is_frozen():
        cmd = [sys.executable, *command_args]
    else:
        cmd = [sys.executable, "-m", "pupil_labs.neon_player", *command_args]

    logger.debug(f"Executing bg job {' '.join(cmd)}")

    self.proc = subprocess.Popen(cmd)  # noqa: S603

    logger.info(f"Background job started: {self.name}")

    def cancel(self) -> None:
    """Stop the background job.

    Once the job is stopped, the ``canceled`` signal is emitted."""
    self.proc.terminate()
    self.proc.wait()
    self.canceled.emit()
